Table role_types {
  id integer [primary key, increment]
  role_name varchar(30) [unique, not null]              // "C"=Cliente, "D"=Dipendente, "A"=Admin
  // Seed: C(1), D(2), A(3)
}

Table user_statuses {
  id integer [primary key, increment]
  user_status varchar(30) [unique, not null]            // PENDING, ACTIVE, ANNULLED, SUSPENDED
  // Seed: PENDING(1), ACTIVE(2), ANNULLED(3), SUSPENDED(4)
}

Table account_statuses {
  id integer [primary key, increment]
  status_name varchar(30) [unique, not null]            // INACTIVE, ACTIVE, FROZEN, CLOSED
  // Seed: INACTIVE(1), ACTIVE(2), FROZEN(3), CLOSED(4)
}

Table card_types {
  id integer [primary key, increment]
  type_name varchar(30) [unique, not null]              // DEBIT
  // Seed: DEBIT(1)
}

Table card_statuses {
  id integer [primary key, increment]
  status_name varchar(30) [unique, not null]            // INACTIVE, ACTIVE, BLOCKED, CLOSED
  // Seed: INACTIVE(1), ACTIVE(2), BLOCKED(3), CLOSED(4)
}

Table limit_types {
  id integer [primary key, increment]
  limit_name varchar(30) [unique, not null]
  notes varchar(255)
  change_policy varchar(20) [not null, enum: USER_FULL, USER_LOWER_ONLY, BANK_ONLY]
  // Seed:
  //   DAILY_TRANSFER(1)       - USER_LOWER_ONLY
  //   SINGLE_TRANSFER(2)      - USER_LOWER_ONLY
  //   INSTANT_TRANSFER_SINGLE(3) - BANK_ONLY
  //   MONTHLY_TRANSFER(4)     - BANK_ONLY
  //   ATM_WITHDRAWAL(5)       - USER_FULL
  //   POS_SPENDING(6)         - USER_FULL
}

Table transaction_types {
  id integer [primary key, increment]
  type_name varchar(30) [unique, not null]              // DEPOSIT, WITHDRAWAL, TRANSFER, INITIAL_TRANSFER, INSTANT_TRANSFER
  // Seed: DEPOSIT(1), WITHDRAWAL(2), TRANSFER(3), INITIAL_TRANSFER(4), INSTANT_TRANSFER(5)
}

Table transaction_statuses {
  id integer [primary key, increment]
  status_name varchar(30) [unique, not null]            // PENDING, COMPLETED, FAILED, REJECTED, CANCELLED
  // Seed: PENDING(1), COMPLETED(2), FAILED(3), REJECTED(4), CANCELLED(5)
}

// ═══════════════════════════════════════════════════════════════════════════════

Table users {
  id bigint [primary key, increment]
  keycloak_id varchar(255) [unique, null]               // FK logica verso Keycloak
  username varchar(50) [unique, not null]               // = email (lowercase)
  first_name varchar(100) [not null]
  last_name varchar(100) [not null]
  birth_date date [null]
  email varchar(150) [unique, not null]                 // lowercase normalizzato
  profession varchar(100) [not null]
  gender varchar(1) [not null]                          // M / F
  fiscal_code varchar(16) [unique, not null]            // 16 caratteri
  phone varchar(20) [not null]
  residence varchar(200) [not null]
  birth_place varchar(100) [not null]
  birth_province varchar(2) [not null]                  // 2 caratteri (es. RM)
  role_type_id integer [not null, ref: > role_types.id]
  status_id integer [not null, ref: > user_statuses.id]
  branch_code varchar(20) [null]
  limits_setup_complete boolean [not null, default: false]
  profile_picture_url varchar(255) [null]
  password_changed_at timestamptz [null]                // usato da JwtPasswordChangeFilter
  created_at timestamptz                                // default: Europe/Rome now()
}

Table accounts {
  id bigint [primary key, increment]
  account_number varchar(50) [unique, not null]         // IT IBAN-like generato
  balance numeric(15,2) [not null, default: 0.00]
  status_id integer [not null]                          // ref logica a account_statuses.id
  user_id bigint [not null, ref: > users.id]
  created_at timestamptz
  closed_at timestamptz [null]
  closure_requested_at timestamptz [null]               // data richiesta chiusura
  closure_rejected_at timestamptz [null]                // data rifiuto chiusura
  source_account_number varchar(50) [null]              // conto origine per apertura con fondi
  initial_amount numeric(15,2) [null]                   // importo iniziale da conto origine
  version bigint [default: 0]                           // @Version ottimistic locking
}

Table account_limits {
  id bigint [primary key, increment]
  account_id bigint [not null, ref: > accounts.id]
  limit_type_id integer [not null, ref: > limit_types.id]
  max_amount numeric(15,2) [not null]
  updated_at timestamptz
  indexes: unique (account_id, limit_type_id)
}

Table cards {
  id bigint [primary key, increment]
  card_number varchar(16) [unique, not null]            // 16 cifre
  holder_name varchar(150) [not null]
  expiration_date date [not null]                       // now() + 5 anni
  cvv varchar(3) [not null]                             // 3 cifre
  card_type_id integer [not null, ref: > card_types.id]
  status_id integer [null, ref: > card_statuses.id]
  account_id bigint [not null]                          // FK logica (nessun constraint JPA)
}

Table transactions {
  id bigint [primary key, increment]
  amount numeric(15,2) [not null]
  type_id integer [not null, ref: > transaction_types.id]
  status_id integer [not null, ref: > transaction_statuses.id]
  description varchar(255) [null]
  source_balance_after numeric(15,2) [null]
  dest_balance_after numeric(15,2) [null]
  created_at timestamptz
  scheduled_date timestamptz [null]                     // data programmata per bonifici non istantanei
  source_account_id bigint [null, ref: > accounts.id]
  destination_account_id bigint [null, ref: > accounts.id]
}

Table beneficiaries {
  id bigint [primary key, increment]
  user_id bigint [not null, ref: > users.id]
  nickname varchar(100) [not null]
  beneficiary_name varchar(150) [null]
  destination_account_number varchar(50) [not null]
  created_at timestamptz
  indexes: unique (user_id, destination_account_number)
}

Table notifications {
  id bigint [primary key, increment]
  user_id bigint [not null, ref: > users.id]            // FK logica (nessun constraint JPA)
  type varchar(50) [not null]                           // ACCOUNT, TRANSFER, DEPOSIT, WITHDRAWAL, SCHEDULED_TRANSFER
  message varchar(500) [not null]
  message_key varchar(100) [null]                       // NOTIF_* per notifiche "importanti"
  message_params varchar(500) [null]                    // JSON array di parametri
  read boolean [not null, default: false]               // column: "read"
  created_at timestamptz [not null]                     // default: Europe/Rome now()
}

Table audit_logs {
  id bigint [primary key, increment]
  entity_type varchar(50) [not null]                    // REGISTRATION, ACCOUNT, etc.
  entity_id bigint [null]
  action varchar(50) [not null]                         // VALIDATE, REJECT, REOPEN, DELETE, ACTIVATE, FREEZE, etc.
  performed_by varchar(100) [not null]                  // "system" o nome admin/employee
  performed_by_user_id bigint [null, ref: > users.id]  // FK logica
  details varchar(500) [null]
  performed_at timestamptz
}

Table user_pins {
  id bigint [primary key, increment]
  user_id bigint [unique, not null, ref: > users.id]    // 1:1 con users
  pin_hash varchar(255) [not null]                      // BCrypt hash
  created_at timestamptz
}

Table password_change_requests {
  id bigint [primary key, increment]
  user_id bigint [not null, ref: > users.id]            // FK logica
  status varchar(20) [not null]                         // PENDING, APPROVED, REJECTED
  created_at timestamptz [not null]                     // default: Europe/Rome now()
  processed_at timestamptz [null]
}

Table limit_change_requests {
  id bigint [primary key, increment]
  user_id bigint [not null]                             // FK logica a users.id
  account_number varchar(30) [not null]
  limit_type_name varchar(30) [not null]                // nome del LimitType
  requested_amount numeric(15,2) [not null]
  current_amount numeric(15,2) [not null]
  status varchar(20) [not null]                         // PENDING, APPROVED, REJECTED
  created_at timestamptz [not null]                     // default: Europe/Rome now()
  processed_at timestamptz [null]
}
